name: Build Windows EXE

on:
  push:
    branches: [ main ]
  workflow_dispatch:

jobs:
  build:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.12'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
          pip install pyinstaller

      - name: Download ffmpeg + ffprobe (juntar vídeo/áudio, gerar MP3 e embutir capa/metadados)
        shell: pwsh
        run: |
          Invoke-WebRequest -Uri "https://www.gyan.dev/ffmpeg/builds/ffmpeg-release-essentials.zip" -OutFile "ffmpeg.zip"
          Expand-Archive -Path "ffmpeg.zip" -DestinationPath "ffmpeg_extracted"
          foreach ($name in @("ffmpeg.exe", "ffprobe.exe")) {
            $f = Get-ChildItem -Path "ffmpeg_extracted" -Recurse -Filter $name | Select-Object -First 1
            Copy-Item $f.FullName -Destination $name
          }
          .\ffmpeg.exe -version
          .\ffprobe.exe -version

      - name: Generate icon
        run: python generate_icon.py

      - name: Build EXE (PyInstaller, arquivo único)
        run: pyinstaller --noconfirm --clean --onefile --windowed --name DXDownloader --icon assets/icon.ico --add-binary "ffmpeg.exe;." --add-binary "ffprobe.exe;." --collect-all yt_dlp --collect-all curl_cffi app.py

      - name: Upload artifact
        uses: actions/upload-artifact@v4
        with:
          name: dx-downloader-windows
          path: dist/DXDownloader.exe
